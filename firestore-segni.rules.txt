rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /menu/{document=**} {
      // Chiunque può leggere il menu (clienti)
      allow read: if true;

      // Solo l'admin loggato può modificare
      allow write: if request.auth != null;
    }

    match /segni/{segnoId} {
      // I visitatori possono soltanto inviare un nuovo disegno.
      allow create: if request.auth == null
        && request.resource.data.keys().hasOnly(['strokes', 'createdAt'])
        && request.resource.data.strokes is list
        && request.resource.data.strokes.size() > 0
        && request.resource.data.strokes.size() <= 200
        && request.resource.data.createdAt is timestamp;

      // I disegni sono leggibili ed eliminabili solo dopo login admin.
      allow read, delete: if request.auth != null;
      allow update: if false;
    }
  }
}
